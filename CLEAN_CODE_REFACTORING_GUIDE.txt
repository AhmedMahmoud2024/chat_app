  class FirebaseRecentsDataSource implements RemoteRecentsDataSource {
    final FirebaseFirestore _firestore;
    
    @override
    Future<List<CallLogModel>> getCallLogs(String userId) async {
      try {
        final snapshot = await _firestore
            .collection('calls')
            .where('userId', isEqualTo: userId)
            .get();
        
        return snapshot.docs
            .map((doc) => CallLogModel.fromJson(doc.data()))
            .toList();
      } on FirebaseException catch (e) {
        throw FirebaseFailure(e.message ?? 'Unknown error');
      }
    }
    // ... deleteCallLog, clearAllLogs
  }

2.2.4 Repository Implementation: recents_repository_impl.dart
  class RecentsRepositoryImpl implements RecentsRepository {
    final RemoteRecentsDataSource remoteDataSource;
    final LocalRecentsDataSource? localDataSource;
    
    @override
    Future<Either<Failure, List<CallLogEntity>>> getCallLogs(String userId) async {
      try {
        // Try remote first
        final models = await remoteDataSource.getCallLogs(userId);
        
        // Cache locally
        if (localDataSource != null) {
          await localDataSource!.saveCallLogs(models);
        }
        
        // Convert models to entities
        final entities = models.map((m) => m.toEntity()).toList();
        return Right(entities);
        
      } on NetworkException catch (e) {
        return Left(NetworkFailure('No internet: ${e.message}'));
      } on FirebaseException catch (e) {
        return Left(FirebaseFailure('Firebase error: ${e.message}'));
      } catch (e) {
        return Left(UnknownFailure('Unexpected error: $e'));
      }
    }
    
    @override
    Future<Either<Failure, bool>> deleteCallLog(String callId) async {
      try {
        await remoteDataSource.deleteCallLog(callId);
        return const Right(true);
      } on FirebaseException catch (e) {
        return Left(FirebaseFailure(e.message ?? 'Delete failed'));
      }
    }
  }
  
  Why?
  - Mediator between domain and data
  - Catches exceptions
  - Wraps in Either<Failure, T>
  - Can implement caching logic

2.3.3 BLoC: recents_bloc.dart
  class RecentsBloc extends Bloc<RecentsEvent, RecentsState> {
    final GetCallLogsUseCase getCallLogsUseCase;
    final DeleteCallLogUseCase deleteCallLogUseCase;
    final ClearAllLogsUseCase clearAllLogsUseCase;
    
    RecentsBloc({
      required this.getCallLogsUseCase,
      required this.deleteCallLogUseCase,
      required this.clearAllLogsUseCase,
    }) : super(const RecentsInitial()) {
      on<FetchRecentsEvent>(_onFetchRecents);
      on<DeleteCallLogEvent>(_onDeleteCallLog);
      on<ClearAllLogsEvent>(_onClearAllLogs);
      on<RefreshRecentsEvent>(_onRefreshRecents);
    }
    
    Future<void> _onFetchRecents(
      FetchRecentsEvent event,
      Emitter<RecentsState> emit,
    ) async {
      emit(const RecentsLoading());
      
      final result = await getCallLogsUseCase.execute(
        GetCallLogsParams(userId: event.userId),
      );
      
      result.fold(
        (failure) => emit(RecentsError(failure.message)),
        (logs) => logs.isEmpty
            ? emit(const RecentsEmpty())
            : emit(RecentsLoaded(logs)),
      );
    }
    
    Future<void> _onDeleteCallLog(
      DeleteCallLogEvent event,
      Emitter<RecentsState> emit,
    ) async {
      final result = await deleteCallLogUseCase.execute(
        DeleteCallLogParams(callId: event.callId),
      );
      
      result.fold(
        (failure) => emit(RecentsError(failure.message)),
        (success) {
          emit(CallLogDeletedSuccess(event.callId));
          // Re-fetch list after delete
          // add(FetchRecentsEvent(currentUserId));
        },
      );
    }
    
    Future<void> _onClearAllLogs(
      ClearAllLogsEvent event,
      Emitter<RecentsState> emit,
    ) async {
      final result = await clearAllLogsUseCase.execute(
        ClearAllLogsParams(userId: event.userId),
      );
      
      result.fold(
        (failure) => emit(RecentsError(failure.message)),
        (success) => emit(const RecentsEmpty()),
      );
    }
    
    Future<void> _onRefreshRecents(
      RefreshRecentsEvent event,
      Emitter<RecentsState> emit,
    ) async {
      // Manual refresh by user
      await _onFetchRecents(FetchRecentsEvent(event.userId), emit);
    }
  }
  
  Why BLoC?
  - Centralized state management
  - Separates business logic from UI
  - Reactive: UI automatically updates
  - Testable: test state transitions independently

    class RecentsScreen extends StatelessWidget {
      final String userId;
      
      const RecentsScreen({required this.userId});
      
      @override
      Widget build(BuildContext context) {
        return BlocProvider(
          create: (context) => RecentsBloc(
            getCallLogsUseCase: getIt<GetCallLogsUseCase>(),
            deleteCallLogUseCase: getIt<DeleteCallLogUseCase>(),
            clearAllLogsUseCase: getIt<ClearAllLogsUseCase>(),
          )..add(FetchRecentsEvent(userId)),
          child: Scaffold(
            appBar: AppBar(
              title: const Text('Recents'),
              actions: [
                IconButton(
                  icon: const Icon(Icons.delete_sweep),
                  onPressed: () {
                    context.read<RecentsBloc>().add(ClearAllLogsEvent(userId));
                  },
                ),
              ],
            ),
            body: BlocBuilder<RecentsBloc, RecentsState>(
              builder: (context, state) {
                if (state is RecentsLoading) {
                  return const Center(child: CircularProgressIndicator());
                }
                
                if (state is RecentsLoaded) {
                  return RefreshIndicator(
                    onRefresh: () async {
                      context.read<RecentsBloc>().add(RefreshRecentsEvent(userId));
                    },
                    child: ListView.builder(
                      itemCount: state.callLogs.length,
                      itemBuilder: (context, index) {
                        return CallLogItemWidget(
                          callLog: state.callLogs[index],
                          onDelete: () {
                            context.read<RecentsBloc>().add(
                              DeleteCallLogEvent(state.callLogs[index].id),
                            );
                          },
                        );
                      },
                    ),
                  );
                }
                
                if (state is RecentsEmpty) {
                  return const EmptyRecentsWidget();
                }
                
                if (state is RecentsError) {
                  return ErrorRecentsWidget(
                    message: state.message,
                    onRetry: () {
                      context.read<RecentsBloc>().add(FetchRecentsEvent(userId));
                    },
                  );
                }
                
                return const SizedBox.shrink();
              },
            ),
          ),
        );
      }
    }
